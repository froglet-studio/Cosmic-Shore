SELECT game_mode,
       COUNT(DISTINCT match_id)      AS matches,
       ROUND(100.0 * AVG(completed), 1) AS finished_pct,
       ROUND(AVG(flight_time_seconds), 1) AS avg_flight
FROM plays
WHERE build_kind = 'release' AND day >= date('now', '-30 days')
  AND play_type = 'solo_vs_ai'
GROUP BY game_mode ORDER BY matches DESC LIMIT 12;
